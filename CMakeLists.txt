  add_compile_definitions(RECOMP_ABI_CHECK=1)

set(XBOXRECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/xboxrecomp" CACHE PATH "xboxrecomp toolkit")

# RECOMP_GEN_DIR / CJ_TARGETS let a second build dir hold a trial target set
# (e.g. build-trial8: work/hyscan/trial8/gen + tools/hybridize/targets-trial8.txt).
set(RECOMP_GEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen" CACHE PATH "tools.recomp output")
set(CJ_TARGETS "${CMAKE_CURRENT_SOURCE_DIR}/tools/hybridize/targets.txt" CACHE FILEPATH "hybridize target list")
file(GLOB RECOMP_GEN_SOURCES CONFIGURE_DEPENDS "${RECOMP_GEN_DIR}/*.c")
# Generated chunks: keep compile time and memory bounded on a small host.
set_source_files_properties(${RECOMP_GEN_SOURCES} PROPERTIES COMPILE_OPTIONS "-O1;-g0;-w;-fno-strict-aliasing;-fwrapv")

# Native swap (checkpoint 1): src/native/native_swap.c is also passed to
# tools.recomp as --exclude-manual; see docs/native-swap.md.
# Hybrid swaps generated from pcj-decomp (docs/hybridize.md). ON: the guest-form
# C++ is regenerated by tools/hybridize on every build from the read-only
# pcj-decomp sources; build/hybrid/hybrid_swap.c is the recompiler's
# --exclude-manual input. OFF: checkpoint 1's hand-copied native_swap.c.
option(CJ_HYBRID_GEN "Native swaps generated by tools/hybridize" ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/pcj-decomp/include")
  set(PCJ_DECOMP_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/pcj-decomp")
else()
  set(PCJ_DECOMP_DEFAULT "/home/claude/pcj-decomp")
endif()
set(PCJ_DECOMP_DIR "${PCJ_DECOMP_DEFAULT}" CACHE PATH "pcj-decomp checkout (read only)")
if(CJ_HYBRID_GEN)
  set(HY_DIR ${CMAKE_BINARY_DIR}/hybrid)
  # the group list is read at configure time: re-configure when it changes (a stale list drops swaps and the link fails)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CJ_TARGETS})
  file(STRINGS ${CJ_TARGETS} HY_LINES REGEX "^[^#]")
  set(HY_SPEC "")
  set(HY_OUT ${HY_DIR}/hybrid_swap.c)
  set(HY_DEPS "")
  foreach(l ${HY_LINES})
    separate_arguments(toks UNIX_COMMAND "${l}")
    list(APPEND HY_SPEC ${toks})
    foreach(t ${toks})
      if(t MATCHES "^unit=(.*)$")
        list(APPEND HY_OUT ${HY_DIR}/hy_${CMAKE_MATCH_1}.cpp)
        file(GLOB u ${PCJ_DECOMP_DIR}/src/*/${CMAKE_MATCH_1}.cpp)
        list(APPEND HY_DEPS ${u})
      endif()
      if(t MATCHES "^mod=(.*)$")
        list(APPEND HY_OUT ${HY_DIR}/hy_${CMAKE_MATCH_1}_mod.cpp)
        file(GLOB u ${PCJ_DECOMP_DIR}/src/*/${CMAKE_MATCH_1}.cpp)
        list(APPEND HY_DEPS ${u})
        if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/mods/${CMAKE_MATCH_1}.cpp)
          list(APPEND HY_DEPS ${CMAKE_CURRENT_SOURCE_DIR}/mods/${CMAKE_MATCH_1}.cpp)
        endif()
      endif()
    endforeach()
  endforeach()
  file(GLOB HY_HDRS ${PCJ_DECOMP_DIR}/include/*.h)
  add_custom_command(OUTPUT ${HY_OUT}
    COMMAND ${CMAKE_COMMAND} -E env PCJ_DECOMP=${PCJ_DECOMP_DIR} python3 ${CMAKE_CURRENT_SOURCE_DIR}/tools/hybridize/hybridize.py gen ${HY_SPEC} -o ${HY_DIR}
    DEPENDS tools/hybridize/hybridize.py ${CJ_TARGETS} ${HY_DEPS} ${HY_HDRS}
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} COMMENT "hybridize: guest-form C++ from pcj-decomp")
  set(SWAP_SOURCES ${HY_OUT} src/native/hybrid/guest_rt.c src/native/hybrid/cjmod.c src/native/hybrid/cjmod_cam.c src/native/hybrid/cjmod_move.c src/native/hybrid/swap_verify.c)
else()
  set(SWAP_SOURCES src/native/native_swap.c src/native/fe_script_native.cpp)
endif()
# CJ_DET_FILE: the gen dir was regenerated with src/det_file.c in the
# --exclude-manual input (sub_00284E60 wrapped as sub_00284E60_gen; RUN.md).
# OFF for an older gen dir (build-trial8): det_file.c then defines no wrapper.
option(CJ_DET_FILE "gen dir wraps FileThreadWakeUp (CJ_DET synchronous file I/O)" ON)
# Decompiled C++ (hybrid swaps) keeps the guest's habits: type punning through
# pointers, wrapping arithmetic, this == NULL checks. GCC and clang exploit the
# UB differently, so turn those optimisations off for both.
set_source_files_properties(${SWAP_SOURCES} PROPERTIES COMPILE_OPTIONS "-fno-strict-aliasing;-fwrapv;-fno-delete-null-pointer-checks")
add_executable(pcj src/main.c src/recomp_manual.c src/determinism.c src/det_file.c src/femouse.c src/native_gfx.c src/native_dsound.c src/mods_archive.c src/cj_rec.c src/gl_backend.c src/gamelog.c src/async_io.c src/frame_pacer.c src/prof.c src/fast_emutls.c src/cj_pad.c src/texpack.c src/textpack.c src/glyphs.c src/keymap.c src/focusdiag.c src/pcsettings.c src/quicksave.c src/cutscene_ai.c ${SWAP_SOURCES} ${RECOMP_GEN_SOURCES})
target_compile_definitions(pcj PRIVATE _GNU_SOURCE)
if(NOT CJ_DET_FILE)
  target_compile_definitions(pcj PRIVATE CJ_NO_FILE_WRAP)
endif()
if(WIN32)
  # MinGW-w64 cross build (docs/windows-build.md): SDL2 + libepoxy from
  # CJ_WIN_DEPS (win-deps/: the SDL2 mingw devel release, epoxy built with meson).
  set(CJ_WIN_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/win-deps" CACHE PATH "Windows SDL2/epoxy prefix")
  target_include_directories(pcj PRIVATE ${CJ_WIN_DEPS}/sdl2/include ${CJ_WIN_DEPS}/epoxy/include)
  target_link_libraries(pcj PRIVATE xboxrecomp
    ${CJ_WIN_DEPS}/sdl2/lib/libSDL2.dll.a ${CJ_WIN_DEPS}/epoxy/lib/libepoxy.dll.a
    opengl32 winmm dbghelp psapi shell32 ole32 pthread)
  target_link_options(pcj PRIVATE -static-libgcc -static-libstdc++ -Wl,--stack,8388608 -Wl,--wrap=xa2_init
    # a normal Windows app: no console window (the log goes to pcj-log.txt; PCJ_LOG=- opens a console)
    -Wl,--subsystem,windows)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CJ_GLDEPS REQUIRED sdl2 epoxy)
  target_include_directories(pcj PRIVATE ${CJ_GLDEPS_INCLUDE_DIRS})
  target_link_libraries(pcj PRIVATE xboxrecomp m pthread dl ${CJ_GLDEPS_LINK_LIBRARIES})
  target_link_options(pcj PRIVATE -rdynamic -no-pie)
endif()

target_include_directories(pcj PRIVATE src src/native src/recomp ${RECOMP_GEN_DIR} ${XBOXRECOMP_DIR}/include)

# Build id recorded in session recordings (src/cj_rec.c): the commit at configure time.
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
                OUTPUT_VARIABLE CJ_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
set_source_files_properties(src/cj_rec.c PROPERTIES COMPILE_DEFINITIONS "CJ_BUILD_ID=\"${CJ_GIT_SHA}\"")

# The same libm on every platform (third_party/openlibm-cj, FreeBSD msun via OpenLibm):
# sin, cos, tan, sincos, atan2, log2, expm1 (the recompiled x87 code) and the float
# versions mods use. glibc, MinGW and Wine each round a few results differently, which
# makes the deterministic mode (CJ_DET, session recordings) diverge between the
# Windows and Linux builds. Linked as objects so they always override the system libm.
file(GLOB CJ_OPENLIBM_SRC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/openlibm-cj/src/*.c)
add_library(cjm OBJECT ${CJ_OPENLIBM_SRC})
target_include_directories(cjm PRIVATE third_party/openlibm-cj/include third_party/openlibm-cj/src)
target_compile_definitions(cjm PRIVATE __BSD_VISIBLE=1)
target_compile_options(cjm PRIVATE -O2 -fno-builtin -ffp-contract=off -fno-fast-math -w)
target_sources(pcj PRIVATE $<TARGET_OBJECTS:cjm>)
